--// ============ HWID ============
local function getHWID()
    local hwid = "UNKNOWN"
    pcall(function()
        if gethwid then
            hwid = gethwid()
        elseif syn and syn.get_hwid then
            hwid = syn.get_hwid()
        elseif KRNL_LOADED and getgenv().KrnlHwid then
            hwid = getgenv().KrnlHwid
        elseif fluxus and fluxus.getHWID then
            hwid = fluxus.getHWID()
        else
            local raw = tostring(LocalPlayer.UserId) .. "_" .. tostring(LocalPlayer.Name) .. "_" .. tostring(game.PlaceId)
            local hash = 0
            for i = 1, #raw do
                hash = (hash * 31 + string.byte(raw, i)) % 2147483647
            end
            hwid = "GEN" .. tostring(hash)
        end
    end)
    return tostring(hwid)
end

local function spawnSparkle()

    sparkle.Size = UDim2.new(0, 20, 0, 20)
    sparkle.Position = UDim2.new(math.random(), 0, 1.1, 0)
    sparkle.Rotation = math.random(0, 360)
    sparkle.ZIndex = 10
    sparkle.Parent = SparkleContainer
    
    local targetPos = UDim2.new(sparkle.Position.X.Scale, sparkle.Position.X.Offset, -0.1, 0)
    local duration = math.random(3, 6)
    
    TweenService:Create(sparkle, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
        Position = targetPos,
        Rotation = sparkle.Rotation + 360,
        TextTransparency = 1,
    }):Play()
    
    task.delay(duration, function()
        if sparkle then sparkle:Destroy() end
    end)
end

--// ============ FUNCTIONS ============
local function setStatus(text, color, dotColor)
    StatusText.Text = text
    StatusText.TextColor3 = color or COLORS.SubText
    if dotColor then
        StatusDot.BackgroundColor3 = dotColor
        StatusDotGlow.BackgroundColor3 = dotColor
    end
end

local function setInfo(line1, line2)
    if line1 then InfoLine1.Text = line1 end
    if line2 then InfoLine2.Text = line2 end
end

local function updateProgressBar()
    if savedData.isPerm then
        ProgressBarBg.Visible = false
        return
    end
    if savedData.key and savedData.expire > 0 then
        ProgressBarBg.Visible = true
        local remaining = savedData.expire - os.time()
        local total = CONFIG.DURATION_12H
        local percent = math.clamp(remaining / total, 0, 1)
        TweenService:Create(ProgressBar, TweenInfo.new(0.5), {
            Size = UDim2.new(percent, 0, 1, 0)
        }):Play()
        
        if percent < 0.2 then
            ProgressGrad.Color = ColorSequence.new(COLORS.Error, COLORS.Warning)
        elseif percent < 0.5 then
            ProgressGrad.Color = ColorSequence.new(COLORS.Warning, COLORS.Gold)
        else
            ProgressGrad.Color = ColorSequence.new(COLORS.Gold, COLORS.Accent, COLORS.Gold)
        end
    end
end

local function formatTime(seconds)
    if seconds <= 0 then return "Het han" end
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = seconds % 60
    if h > 0 then
        return string.format("%dh %dm", h, m)
    elseif m > 0 then
        return string.format("%dm %ds", m, s)
    else
        return string.format("%ds", s)
    end
end

local function validateKey(key)
    key = key:gsub("%s+", ""):upper()
    
    if CONFIG.KEYS_PERM[key] then
        return true, "perm", CONFIG.KEYS_PERM[key].note
    elseif CONFIG.KEYS_12H[key] then
        return true, "12h", CONFIG.KEYS_12H[key].note
    end
    return false, nil, nil
end

--// ============ ANIMATION: CLOSE MENU ============
local function closeMenuAnimation(callback)
    -- 1. Hien thi thong bao thanh cong
    setStatus("Tai script thanh cong!", COLORS.Success, COLORS.Success)
    setInfo("Trang thai: Da kich hoat", "Loai key: OK")
    
    task.wait(0.6)
    
    -- 2. Phat sang toan bo menu
    local flashFrame = Instance.new("Frame")
    flashFrame.Size = UDim2.new(1, 0, 1, 0)
    flashFrame.BackgroundColor3 = COLORS.Success
    flashFrame.BackgroundTransparency = 0.6
    flashFrame.BorderSizePixel = 0
    flashFrame.ZIndex = 999
    flashFrame.Parent = Main
    Instance.new("UICorner", flashFrame).CornerRadius = UDim.new(0, 20)
    
    TweenService:Create(flashFrame, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
    task.wait(0.15)
    
    -- 3. Fade tat ca
    local fadeInfo = TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    
    TweenService:Create(Main, fadeInfo, {
        Size = UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 1
    }):Play()
    
    TweenService:Create(MainStroke, fadeInfo, {Transparency = 1}):Play()
    TweenService:Create(GlowFrame, fadeInfo, {BackgroundTransparency = 1}):Play()
    TweenService:Create(GlowFrame2, fadeInfo, {BackgroundTransparency = 1}):Play()
    TweenService:Create(GlowFrame3, fadeInfo, {BackgroundTransparency = 1}):Play()
    
    for _, v in pairs(Main:GetDescendants()) do
        if v:IsA("TextLabel") or v:IsA("TextButton") or v:IsA("TextBox") then
            TweenService:Create(v, fadeInfo, {TextTransparency = 1, BackgroundTransparency = 1}):Play()
        elseif v:IsA("Frame") then
            pcall(function()
                TweenService:Create(v, fadeInfo, {BackgroundTransparency = 1}):Play()
            end)
        elseif v:IsA("UIStroke") then
            TweenService:Create(v, fadeInfo, {Transparency = 1}):Play()
        end
    end
    
    task.wait(0.6)
    
    -- 4. Destroy GUI
    if ScreenGui then
        pcall(function() ScreenGui:Destroy() end)
    end
    
    if callback then callback() end
end

local function runMainScript()
    setStatus("Dang tai script...", COLORS.Gold, COLORS.Gold)
    task.wait(0.5)
    
    local ok, err = pcall(function()
        loadstring(game:HttpGet(CONFIG.SCRIPT_URL))()
    end)
    
    if ok then
        closeMenuAnimation(function()
            print("[SelomenHub V6] Menu closed - Script loaded OK")
        end)
    else
        setStatus("Loi: " .. tostring(err), COLORS.Error, COLORS.Error)
        setInfo("Trang thai: Loi tai script", "Loai key: --")
    end
end

local function checkSavedKey()
    if savedData.key and not savedData.isPerm then
        if CONFIG.ENABLE_12H_HWID_LOCK and savedData.hwid and savedData.hwid ~= CURRENT_HWID then
            setStatus("Key 12H da dung tren may khac!", COLORS.Error, COLORS.Error)
            setInfo("Trang thai: Bi khoa HWID", "Loai key: 12H")
            return false
        end
    end
    
    if savedData.isPerm then
        setInfo("Trang thai: Da kich hoat (Vinh vien)", "Loai key: PERMANENT")
        return true
    end
    
    if savedData.key and savedData.expire > os.time() then
        local remaining = savedData.expire - os.time()
        setInfo("Trang thai: Con " .. formatTime(remaining), "Loai key: 12H (HWID Lock)")
        updateProgressBar()
        return true
    end
    
    if savedData.key and savedData.expire <= os.time() then
        setStatus("Key da het han. Nhap key moi!", COLORS.Error, COLORS.Error)
        savedData.key = nil
        savedData.expire = 0
        saveData(savedData)
    end
    
    return false
end

CheckBtn.MouseButton1Click:Connect(function()
    if savedData.lockUntil and savedData.lockUntil > os.time() then
        local wait = savedData.lockUntil - os.time()
        setStatus("Bi khoa " .. wait .. "s!", COLORS.Error, COLORS.Error)
        return
    end
    
    local key = KeyInput.Text
    if key == "" then
        setStatus("Vui long nhap key!", COLORS.Error, COLORS.Error)
        return
    end
    
    local upperKey = key:gsub("%s+", ""):upper()
    if not upperKey:match("^QBAO%-HUB%-") then
        setStatus("Key phai bat dau bang QBAO-HUB-!", COLORS.Error, COLORS.Error)
        return
    end
    
    setStatus("Dang kiem tra key...", COLORS.Gold, COLORS.Gold)
    task.wait(0.5)
    
    local valid, keyType, note = validateKey(key)
    
    if not valid then
        savedData.failedAttempts = (savedData.failedAttempts or 0) + 1
        
        if savedData.failedAttempts >= CONFIG.MAX_FAILED_ATTEMPTS then
            savedData.lockUntil = os.time() + CONFIG.LOCKOUT_TIME
            savedData.failedAttempts = 0
            saveData(savedData)
            setStatus("Nhap sai qua nhieu! Khoa " .. CONFIG.LOCKOUT_TIME .. "s", COLORS.Error, COLORS.Error)
        else
            saveData(savedData)
            local remain = CONFIG.MAX_FAILED_ATTEMPTS - savedData.failedAttempts
            setStatus("Key sai! Con " .. remain .. " lan thu", COLORS.Error, COLORS.Error)
        end
        return
    end
    
    savedData.failedAttempts = 0
    
    if keyType == "perm" then
        savedData.key = upperKey
        savedData.expire = 0
        savedData.isPerm = true
        savedData.hwid = nil
        savedData.activatedAt = os.time()
        saveData(savedData)
        
        setStatus("Key VINH VIEN hop le!", COLORS.Success, COLORS.Success)
        setInfo("Trang thai: Da kich hoat (Vinh vien)", "Loai key: " .. (note or "PERMANENT"))
        task.wait(0.7)
        runMainScript()
        
    elseif keyType == "12h" then
        savedData.key = upperKey
        savedData.expire = os.time() + CONFIG.DURATION_12H
        savedData.isPerm = false
        savedData.hwid = CURRENT_HWID
        savedData.activatedAt = os.time()
        saveData(savedData)
        
        setStatus("Key 12H hop le!", COLORS.Success, COLORS.Success)
        setInfo("Trang thai: Con 12h", "Loai key: " .. (note or "12H") .. " (HWID Lock)")
        task.wait(0.7)
        runMainScript()
    end
end)
